```c
static void Task_LobbyRenderRight(u8 taskId)
{
    u16 species, personality;
    u8  slot = sLobbyState.selectedSlot;  // 0 on first entry

    // Print player name to right panel (window 1).
    // ConvertInternationalString converts the encoded playerName to displayable text.
    {
        u8 nameBuf[PLAYER_NAME_LENGTH + 1];
        StringCopy(nameBuf, gSaveBlock2Ptr->playerName);
        AddTextPrinterParameterized4(1, FONT_NORMAL, 4, 4, 0, 0, nameBuf, TEXT_SKIP_DRAW, NULL);
        CopyWindowToVram(1, COPYWIN_GFX);
    }

    // Create animated party icon for selected slot.
    species     = GetMonData(&gPlayerParty[slot], MON_DATA_SPECIES, NULL);
    personality = GetMonData(&gPlayerParty[slot], MON_DATA_PERSONALITY, NULL);

    LoadMonIconPalette(species);
    // Icon placed at right-half center: screen x=200, y=72 (right panel center ~x=180+20)
    sLobbyState.iconSpriteId = CreateMonIcon(species, SpriteCB_MonIcon, 200, 72, 4, personality);

    gTasks[taskId].func = Task_LobbyMain;
}
```

`StringCopy` is from `string_util.h` (already included in Plan 06-02). `GetMonData` and `gPlayerParty` are from `pokemon.h` (already included). `LoadMonIconPalette`, `CreateMonIcon`, `SpriteCB_MonIcon` are from `pokemon_icon.h` (included in Plan 06-02).

Note: `gSaveBlock2Ptr->playerName` is already an encoded string — pass it directly to `AddTextPrinterParameterized4`; the text engine handles decoding.

```c
static void Task_LobbyChangePokemon(u8 taskId)
{
    // Destroy party icon before leaving (sprites are wiped on CB2 re-entry anyway,
    // but clean up cleanly in case this is called mid-session).
    FreeAndDestroyMonIconSprite(&gSprites[sLobbyState.iconSpriteId]);
    FreeMonIconPalette(GetMonData(&gPlayerParty[sLobbyState.selectedSlot], MON_DATA_SPECIES, NULL));

    // Signal that CB2_DenLobbyScreen should handle re-entry differently.
    sLobbyState.returnedFromParty = TRUE;

    // Fade out then open party screen.
    BeginNormalPaletteFade(PALETTES_ALL, 0, 0, 16, RGB_BLACK);
    gTasks[taskId].func = Task_LobbyChangePokemon_WaitFade;
}

static void Task_LobbyChangePokemon_WaitFade(u8 taskId)
{
    if (!gPaletteFade.active)
    {
        // ChooseMonForTradingBoard opens party screen with PARTY_ACTION_CHOOSE_MON
        // and returns to CB2_DenLobbyScreen when the player selects or cancels.
        ChooseMonForTradingBoard(PARTY_MENU_TYPE_FIELD, CB2_DenLobbyScreen);
        DestroyTask(taskId);
    }
}
```

```c
void CB2_DenLobbyScreen(void)
{
    if (sLobbyState.returnedFromParty)
    {
        u8 selected = GetCursorSelectionMonId();
        if (selected < PARTY_SIZE)
            sLobbyState.selectedSlot = selected;
        sLobbyState.returnedFromParty = FALSE;
    }
    else
    {
        sLobbyState.denId         = (u8)gSpecialVar_0x8000;
        sLobbyState.selectedSlot  = 0;
    }

    // ... rest of existing CB2_DenLobbyScreen init unchanged ...
}
```

The full VRAM/OAM/sprite reset that follows is intentional — all sprites are recreated each entry. `sLobbyState` survives because it is EWRAM_DATA.
